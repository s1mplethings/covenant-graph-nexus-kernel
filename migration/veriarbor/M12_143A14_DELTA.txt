+def _decode_media_payload(request):
+    """Decode base64 media bytes from a media.import request.
+
+    Returns (file_bytes, filename, kind). Raises ValueError on invalid input.
+    """
+    data = request.get('data')
+    if not isinstance(data, str) or not data:
+        raise ValueError('data must be a non-empty base64 string')
+
+    # Tolerate data URLs (data:<mime>;base64,<payload>)
+    if data.startswith('data:') and ',' in data:
+        data = data.split(',', 1)[1]
+
+    try:
+        file_bytes = base64.b64decode(data, validate=True)
+    except (binascii.Error, ValueError):
+        raise ValueError('data is not valid base64')
+
+    if not file_bytes:
+        raise ValueError('decoded media is empty')
+
+    filename = request.get('filename')
+    if not isinstance(filename, str) or not filename:
+        filename = 'upload.bin'
+    # Strip any path components to keep the temp file name safe.
+    filename = os.path.basename(filename) or 'upload.bin'
+
+    kind = request.get('kind')
+    if not isinstance(kind, str) or not kind:
+        raise ValueError('kind must be a non-empty string')
+
+    return file_bytes, filename, kind
+

+def _import_media_from_base64(data_dir, request):
+    """Adapter: decode base64 payload, stage to a bounded temp file, call M11."""
+    if m11_handle is None:
+        raise ValueError('M11 module unavailable')
+
+    file_bytes, filename, kind = _decode_media_payload(request)
+
+    _, ext = os.path.splitext(filename)
+    if not ext:
+        ext = '.bin'
+
+    tmp_dir = os.path.join(data_dir, 'tmp')
+    os.makedirs(tmp_dir, exist_ok=True)
+
+    fd, tmp_path = tempfile.mkstemp(prefix='media_import_', suffix=ext, dir=tmp_dir)
+    try:
+        with os.fdopen(fd, 'wb') as fh:
+            fh.write(file_bytes)
+
+        m11_request = {
+            'milestone': 'M11',
+            'op': 'media.import',
+            'project_id': request.get('project_id'),
+            'path': tmp_path,
+            'kind': kind,
+        }
+        return m11_handle(data_dir, m11_request)
+    finally:
+        try:
+            if os.path.exists(tmp_path):
+                os.remove(tmp_path)
+        except OSError:
+            pass
