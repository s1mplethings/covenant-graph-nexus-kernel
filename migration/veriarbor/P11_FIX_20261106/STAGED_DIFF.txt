+"""Read-only verification of FileRunStore facts, not a second acceptance authority.
+
+The trusted coordinator freezes bindings before execution. Candidate/model code
+must not have write access to the run store. Hashes detect changed references;
+they are not signatures against an attacker with host filesystem privileges.
+"""

+def file_hash(path):
+    return hashlib.sha256(Path(path).read_bytes()).hexdigest()
+
+
+def reference(root, path):
+    path = Path(path).resolve()
+    return {'path':path.relative_to(Path(root).resolve()).as_posix(), 'sha256':file_hash(path)}

+def put_evidence(store, project_id, run_id, value):
+    """Idempotent typed immutable documents inside the existing run store."""
+    path = store.run_dir(project_id, run_id)/'evidence'/(identity(value)+'.json')
+    store.immutable_json(path, value)
+    return reference(store.root, path)

+def freeze_goal_bindings(store, project_id, run_id, plan):
+    """Append bindings for admitted goals under a pre-frozen checker policy.
+
+    Existing fixed-goal contracts are unchanged. These documents are evidence
+    in FileRunStore, not a second plan or acceptance authority.
+    """
+    root=store.run_dir(project_id,run_id)
+    path=root/'provenance_contract.json'
+    if not path.exists(): return
+    base=json.loads(path.read_text(encoding='utf-8'))
+    templates=base.get('incremental_checker_bindings')
+    if templates is None: return
+    for goal in plan['goals']:
+        if goal['checker'] not in templates:
+            raise ValueError('goal checker is outside frozen provenance policy')
+        if any(r not in base['requirements'] for r in goal['requirement_refs']):
+            raise ValueError('goal requirements are outside frozen provenance policy')
+        row={'kind':'IncrementalGoalBinding','goal_id':goal['id'],
+             'requirement_refs':goal['requirement_refs'],'subject_refs':goal['subject_refs'],
+             'checker_key':goal['checker'],'contract':{**templates[goal['checker']],
+                 'condition_hash':identity(goal['acceptance'])}}
+        store.immutable_json(root/'provenance_goals'/(identity(goal['id'])+'.json'),row)

+class ProvenanceError(ValueError):
+    def __init__(self, status, reason):
+        self.status, self.reason = status, reason

+class ProvenanceGate:
+    def __init__(self, store, project_id, run_id, *, release_trust_anchor=None,
+                 release_authorization=None):
+        self.store, self.project_id, self.run_id = store, project_id, run_id
+        self.root = store.run_dir(project_id, run_id)
+        # The public key must live outside agent-writable roots.  The signed
+        # authorization may live in the run store because its bytes are signed.
+        self.release_trust_anchor = (Path(release_trust_anchor).resolve()
+                                     if release_trust_anchor else None)
+        self.release_authorization_path = (Path(release_authorization).resolve()
+                                           if release_authorization

+    def resolve(self, ref, kind=None):
+        if not isinstance(ref, dict) or not ref.get('path') or not ref.get('sha256'):
+            raise ProvenanceError('INSUFFICIENT_EVIDENCE', 'missing reference identity')
+        relative = Path(ref['path'])
+        path = self.store.root / relative
+        if relative.is_absolute() or '..' in relative.parts or not path.resolve().is_relative_to(self.store.root):
+            raise ProvenanceError('INVALID_PROVENANCE', 'reference outside this run')
+        if not path.resolve().is_relative_to(self.root):
+            binding=self.binding()
+            if ref not in binding.get('authorized_imports',[]):
+                raise ProvenanceError('INVALID_PROVENANCE','cross-run reference is not explicitly frozen for this logical task')
+        if any(p.is_symlink() for p in [path, *path.parents] if p.is_relative_to(self.store.root)):
+            raise ProvenanceError('INVALID_PROVENANCE', 'symlink evidence')
+        if not path.is_file():
+            raise ProvenanceError('INSUFFICIENT_EVIDENCE', 'referenced object missing: '+relative.as_posix())
+        if file_hash(path) != ref['sha256']:
+            raise ProvenanceError('INVALID_PROVENANCE', 'object hash mismatch: '+relative.as_posix())
+        if kind is None:
+            return path.read_bytes()
+        try:
+            obj = json.loads(path.read_text(encoding='utf-8'))
+        except (ValueError, UnicodeError) as exc:
+            raise ProvenanceError('INVALID_PROVENANCE', 'invalid JSON object') from exc
+        if not isinstance(obj, dict) or obj.get('kind') != kind:
+            raise ProvenanceError('INVALID_PROVENANCE', 'object type mismatch: '+kind)
+        return obj

+    @contextmanager
+    def read_binding_snapshot(self):
+        """One read operation, no cache across views or evidence verification.
+
+        Revalidate the full binding before returning any projected verdict.
+        Individual receipts, referenced objects and current bytes remain checked.
+        """
+        if getattr(self, '_binding_snapshot', None) is not None:
+            raise ProvenanceError('INVALID_PROVENANCE', 'nested binding snapshot')
+        try:
+            before = self._load_binding()
+        except (OSError, ValueError, KeyError, TypeError) as exc:
+            raise ProvenanceError('INVALID_PROVENANCE', 'binding snapshot unavailable') from exc
+        self._binding_snapshot = before
+        try:
+            yield self
+            try:
+                after = self._load_binding()
+            except (OSError, ValueError, KeyError, TypeError) as exc:
+                raise ProvenanceError('INVALID_PROVENANCE', 'binding changed during read') from exc
+            if after != before:
+                raise ProvenanceError('INVALID_PROVENANCE', 'binding changed during read')
+        finally:
+            self._binding_snapshot = None
+
+    def binding(self):
+        snapshot = getattr(self, '_binding_snapshot', None)
+        return deepcopy(snapshot) if snapshot is not None else self._load_binding()

+    def _load_binding(self):
+        path = self.root/'provenance_contract.json'
+        if not path.exists():
+            raise ProvenanceError('INSUFFICIENT_EVIDENCE','no coordinator-frozen provenance contract')
+        obj = json.loads(path.read_text(encoding='utf-8'))
+        if obj.get('kind') != 'ProvenanceContract':
+            raise ProvenanceError('INVALID_PROVENANCE','invalid provenance contract type')
+        if 'incremental_checker_bindings' in obj:
+            obj['goals']=dict(obj['goals'])
+            obj['current_artifacts']=dict(obj.get('current_artifacts',{}))
+            workspace=Path(obj['workspace_root']).resolve()
+            for path in sorted((self.root/'provenance_goals').glob('*.json')):
+                row=json.loads(path.read_text(encoding='utf-8'))
+                if (row.get('kind')!='IncrementalGoalBinding'
+                        or path.stem!=identity(row['goal_id'])
+                        or row['goal_id'] in obj['goals']
+                        or any(r not in obj['requirements'] for r in row['requirement_refs'])):
+                    raise ProvenanceError('INVALID_PROVENANCE','invalid incremental goal identity')
+                template=obj['incremental_checker_bindings'][row['checker_key']]
+                if {k:v for k,v in row['contract'].items() if k!='condition_hash'}!=template:
+                    raise ProvenanceError('INVALID_PROVENANCE','incremental checker policy changed')
+                obj['goals'][row['goal_id']]=row['contract']
+                for subject in row['subject_refs']:
+                    candidate=(workspace/subject).resolve()
+                    if not candidate.is_relative_to(workspace):
+                        raise ProvenanceError('INVALID_PROVENANCE','incremental artifact escape')
+                    obj['current_artifacts'][subject]=str(candidate)
+        from .checker_revision import load_checker_revisions
+        revisions, _ = load_checker_revisions(self, obj)
+        if revisions:
+            obj['checker_revisions'] = revisions
+        return obj

+    def current(self, acceptance):
+        """Resolve the same authoritative acceptance records; never grant a verdict."""
+        view = acceptance.view()
+        binding = self.binding()
+        if binding['requirements'] != acceptance.requirements:
+            raise ProvenanceError('INVALID_PROVENANCE','original requirements differ')
+        chains=[]
+        for goal in view['goals']:
+            evidence=goal.get('evidence')
+            if not evidence:
+                raise ProvenanceError('INSUFFICIENT_EVIDENCE','goal has no current execution')
+            path=self.root/'attempts'/evidence['node_id']/(evidence['attempt_id']+'.json')
+            from cgkn.execution import BlockExecutionResult
+            if not path.exists():raise ProvenanceError('INSUFFICIENT_EVIDENCE','missing attempt receipt')
+            value=json.loads(path.read_text(encoding='utf-8'))
+            if identity(value)!=evidence['receipt_hash']:
+                raise ProvenanceError('INVALID_PROVENANCE','acceptance receipt hash mismatch')
+            result=BlockExecutionResult.from_dict(value)
+            current_subjects={k:view['subject_identities'][k] for k in goal['subject_refs']}
+            effective_subjects=evidence.get('effective_subjects')
+            if evidence.get('certified_substitutions'):
+                from cgkn.adapters.veriarbor.substitution_adapter import replacements, reuse_cover
+                with self.read_binding_snapshot():
+                    proofs, checkers = replacements(acceptance, self)
+                    cover = reuse_cover(goal, evidence, current_subjects, proofs, checkers)
+                if not cover or cover != evidence['certified_substitutions'] or effective_subjects != current_subjects:
+                    raise ProvenanceError('STALE','certified replacement no longer covers current dependencies')
+                report=self.verify_check(result,goal,evidence['subjects'])
+                if report['status']=='TRUSTED':
+                    report['chain']={**report['chain'],'certified_substitutions':cover,
+                                     'effective_subjects':current_subjects}
+            else:
+                report=self.verify_check(result,goal,current_subjects)
+            if report['status']!='TRUSTED':raise ProvenanceError(report['status'],report['reason'])
+            chains.append(report['chain'])
+        if not chains:raise ProvenanceError('INSUFFICIENT_EVIDENCE','no current checks')
+        return {'status':'TRUSTED','acceptance':view,'chains':chains}

+"""Signed release-authorization verification for non-Worker delivery sources.
+
+Trust model
+-----------
+The run store and the candidate workspace are writable by the executing agent.
+Nothing inside them can prove *who* authorized a release.  This module only
+accepts a release authorization that is signed with Ed25519 by a key listed in
+a trust anchor file that lives outside both writable roots.  The verifier never
+reads a private key.
+
+The trust anchor file still has to be protected by the host (filesystem ACL,
+separate OS account, HSM, or a remote signing service).  If the agent can write
+the anchor or invoke the signer, this module provides tamper *evidence* only,
+not a real trust boundary; callers must report PENDING_TRUST_ANCHOR in that
+case.  See docs/RELEASE_AUTHORIZATION.md for the deployment contract.
+"""
+from __future__ import annotations
+
+import base64
+import hashlib
+import json
+from datetime import datetime, timezone
+from pathlib import Path
+
+AUTHORIZATION_KIND = "SignedReleaseAuthorization"
+ANCHOR_KIND = "ReleaseTrustAnchor"
+SCHEMA_VERSION = "1"
+def _b64(value, validate=False):
+    return base64.b64decode(value, validate=validate)
+
+
+class ReleaseAuthorizationError(ValueError):
+    """Fail-closed authorization error; never downgraded to a warning."""
+
+
+def _canonical(value):
+    return json.dumps(value, sort_keys=True, ensure_ascii=False,
+                      separators=(",", ":"), allow_nan=False)
+
+
+def payload_bytes(document):
+    """Canonical bytes covered by the signature (everything except signature)."""
+    if not isinstance(document, dict):
+        raise ReleaseAuthorizationError("authorization must be an object")
+    signed = {key: value for key, value in document.items() if key != "signature"}
+    return _canonical(signed).encode("utf-8")
+
+
+def manifest_digest(manifest):
+    normalized = normalize_manifest(manifest)
+    return hashlib.sha256(_canonical(normalized).encode("utf-8")).hexdigest()
+
+
+def normalize_manifest(manifest):
+    if not isinstance(manifest, dict) or not manifest:
+        raise ReleaseAuthorizationError("delivery manifest must be a non-empty object")
+    normalized = {}
+    for name, value in sorted(manifest.items()):
+        if not isinstance(name, str) or not name:
+            raise ReleaseAuthorizationError("manifest artifact name is invalid")
+        digest = str(value).removeprefix("sha256:").lower()
+        if len(digest) != 64 or any(ch not in "0123456789abcdef" for ch in digest):
+            raise ReleaseAuthorizationError("manifest artifact hash is not SHA-256: " + name)
+        normalized[name] = digest
+    return normalized

+def _parse_time(value, field):

+        raise ReleaseAuthorizationError("authorization " + field + " is missing")
+    text = value.strip().replace("Z", "+00:00")
+    try:
+        parsed = datetime.fromisoformat(text)
+    except ValueError as exc:
+        raise ReleaseAuthorizationError("authorization " + field + " is not ISO-8601") from exc
+    if parsed.tzinfo is None:
+        raise ReleaseAuthorizationError("authorization " + field + " needs a timezone")
+    return parsed.astimezone(timezone.utc)

+def _outside(path, root, label):

+        if path.is_relative_to(root):

+    except OSError as exc:

+def load_trust_anchor(anchor_path, *, store_root, workspace_root):
+    """Load the protected public-key configuration; never a private key."""
+    try:
+        anchor = Path(anchor_path).resolve(strict=True)
+    except (OSError, RuntimeError) as exc:
+        raise ReleaseAuthorizationError("release trust anchor is missing") from exc
+    if anchor.is_symlink():

+            or not document.get("authority_id") or not isinstance(document.get("keys"), dict)):

+            or signature.get("key_id") != key_id or not signature.get("value_b64")):
+        raise ReleaseAuthorizationError("authorization signature block is invalid")

+    except Exception as exc:

+    if moment.astimezone(timezone.utc) < not_before:

+    if not document.get("scope"):

+        "trust_anchor_path": anchor["_path"],
+        "signer_key": {key: value for key, value in anchor["keys"][key_id].items()
+                       if key != "public_key_b64"},

+    }

+"""Sign one batch-scoped release authorization with a protected Ed25519 key.
+
+Run this OUTSIDE the agent workspace, from a controller/account that can read
+the private key.  The verifier never calls this module and never needs the
+private key.  Do not check a production private key into the repository and do
+not store it in the run store or candidate workspace.

+def _sha(path):
+    return hashlib.sha256(pathlib.Path(path).read_bytes()).hexdigest()
+
+
+def _read(path):
+    return json.loads(pathlib.Path(path).read_text(encoding="utf-8"))
+
+
+def _resolve(store_root, ref):
+    root = pathlib.Path(store_root).resolve()
+    path = (root / ref["path"]).resolve()
+    if not path.is_relative_to(root) or not path.is_file():
+        raise SystemExit("referenced object is outside the run store: " + str(path))
+    if _sha(path) != ref["sha256"]:
+        raise SystemExit("referenced object hash changed: " + str(path))
+    return path
+

+def build_artifacts(*, store_root, manifest, source_imports):
+    artifacts = {}
+    for path in sorted(pathlib.Path(source_imports).glob("*.json")):
+        doc = _read(path)
+        name = doc.get("artifact")
+        if name not in manifest:
+            continue
+        if doc.get("sha256") != manifest[name]:
+            raise SystemExit("source import does not match the delivery manifest: " + str(name))
+        if doc.get("kind") == "AuthorizedAgentSource":
+            auth = _resolve(store_root, doc["authorization_ref"])
+            review_path = _resolve(store_root, doc["review_ref"])
+            review = _read(review_path)
+            b10_path = _resolve(store_root, review["b10_certificate_ref"])
+            b11_path = _resolve(store_root, review["b11_acceptance_ref"])
+            b10 = _read(b10_path)
+            sections = b10.get("relevant_section_hashes", {})
+            artifacts[name] = {
+                "sha256": manifest[name],
+                "author_type": "agent",
+                "author_identity": doc.get("author_identity", ""),
+                "authorization_sha256": _sha(auth),
+                "review_sha256": _sha(review_path),
+                "b10_certificate_sha256": _sha(b10_path),
+                "b11_acceptance_sha256": _sha(b11_path),
+                "b10_section_sha256": sections.get(name + "#full"),
+            }
+        elif doc.get("kind") == "ImportedWorkerSource":
+            artifacts[name] = {
+                "sha256": manifest[name],
+                "author_type": "worker",
+                "receipt_sha256": doc.get("receipt_ref", {}).get("sha256"),
+                "source_project_id": doc.get("source_project_id"),
+                "source_run_id": doc.get("source_run_id"),
+            }
+        else:
+            raise SystemExit("unknown source import kind: " + str(doc.get("kind")))
+    missing = sorted(set(manifest) - set(artifacts))
+    if missing:
+        raise SystemExit("manifest artifacts without source imports: " + ",".join(missing))
+    return artifacts
